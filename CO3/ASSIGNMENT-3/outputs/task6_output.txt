[1] The 10 definite-clause rules
   R1  Superior base                                Superior(?m,?e)  <=  ReportsTo(?e,?m)
   R2  Superior (right-recursive)                   Superior(?m,?e)  <=  ReportsTo(?e,?y) ∧ Superior(?m,?y)
   R3  Peer                                         Peer(?x,?y)  <=  ReportsTo(?x,?m) ∧ ReportsTo(?y,?m) ∧ neq(?x,?y)
   R4  SameDept                                     SameDept(?x,?y)  <=  InDept(?x,?d) ∧ InDept(?y,?d) ∧ neq(?x,?y)
   R5  Approver (small request: direct manager)     Approver(?m,?r)  <=  RequestBy(?r,?e) ∧ ReportsTo(?e,?m) ∧ Small(?r)
   R6  Approver (large request: a director above)   Approver(?m,?r)  <=  RequestBy(?r,?e) ∧ Superior(?m,?e) ∧ Large(?r) ∧ Director(?m)
   R7  Authorised                                   Authorised(?m,?r)  <=  Approver(?m,?r) ∧ Active(?m)
   R8  Escalate                                     Escalate(?r,?m)  <=  RequestBy(?r,?e) ∧ Superior(?m,?e)
   R9  Delegate                                     Delegate(?x,?y)  <=  Peer(?x,?y) ∧ SameDept(?x,?y) ∧ Active(?y)
   R10 Chain-to-director                            Chain(?e,?m)  <=  Superior(?m,?e) ∧ Director(?m)
   R2L Superior (LEFT-recursive)                    Superior(?m,?e)  <=  Superior(?y,?e) ∧ ReportsTo(?y,?m)    (replaces R2 in Part B)
